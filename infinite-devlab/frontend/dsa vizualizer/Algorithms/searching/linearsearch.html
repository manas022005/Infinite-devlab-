int linearSearch(vector<int>& arr, int target) {
  int n = arr.size();
  for (int i = 0; i < n; i++) { // Walk left to right
    if (arr[i] == target) { // Compare
      return i; // Found at index i
    }
  }
  return -1; // Element not found in array
}
