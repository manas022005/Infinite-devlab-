BINARY SEARCH

Efficiently finds an element in a sorted array by repeatedly halving the search interval. Time Complexity: O(log n).

// Array Sorted. Enter target.
int binarySearch(vector<int>& arr, int target) {
  int low = 0, high = arr.size() - 1;
  while (low <= high) {
    int mid = low + (high - low) / 2;
    if (arr[mid] == target) return mid;
    if (arr[mid] < target) low = mid + 1;
    else high = mid - 1;
  }
  return -1;
}
