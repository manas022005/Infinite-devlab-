SELECTION SORT

Repeatedly finds the minimum element from the unsorted part and swaps it into the target boundary slot. Makes O(n²) comparisons but at most O(n) swaps.

void selectionSort(vector<int>& arr) {
  int n = arr.size();
  for (int i = 0; i < n - 1; i++) {
    int minIdx = i; // Assume target is min initially
    for (int j = i + 1; j < n; j++) {
      if (arr[j] < arr[minIdx]) {
        minIdx = j; // New minimum found
      }
    }
    if (minIdx != i) {
      swap(arr[i], arr[minIdx]); // Place min at target
    }
  }
}
