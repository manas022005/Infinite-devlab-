void dijkstra(int src) {   vector<int> dist(V, INT_MAX);   priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> pq;   dist[src] = 0;   pq.push({0, src});   while (!pq.empty()) {     auto [d, u] = pq.top(); pq.pop();
    if (d > dist[u]) continue; // Skip stale     for (auto& edge : graph[u]) { // Check neighbors       int v = edge.to, weight = edge.wt;       if (dist[u] + weight < dist[v]) {         dist[v] = dist[u] + weight; // Relax         pq.push({dist[v], v});       }     }   } }
