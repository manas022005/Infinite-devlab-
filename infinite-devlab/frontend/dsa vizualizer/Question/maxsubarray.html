// Array generated. Ready to calculate max subarray.
int maxSubArray(vector<int>& arr) {
  int max_sum = INT_MIN;
  int curr_sum = 0;
  for (int i = 0; i < arr.size(); i++) {
    // Extend window or start fresh from arr[i]
    curr_sum = max(arr[i], curr_sum + arr[i]);
    // Update overall best
    if (curr_sum > max_sum) {
      max_sum = curr_sum;
    }
  }
  return max_sum;
}
