void multiply(vector<vector<int>>& A, vector<vector<int>>& B, vector<vector<int>>& C) {
  int n = A.size();
  for (int i = 0; i < n; i++) {        // Rows of A
    for (int j = 0; j < n; j++) {      // Cols of B
      C[i][j] = 0;
      for (int k = 0; k < n; k++) {    // Dot product
        C[i][j] += A[i][k] * B[k][j];
      }
    }
  }
}
