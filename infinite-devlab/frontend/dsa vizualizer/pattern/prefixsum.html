The Math
Sum(L, R) = P[R] - P[L-1]

We take the total sum from index 0 to R, and subtract the sum of the elements before L.

P[] Prefix Sum Array
// Build the prefix array first...
// Phase 1: Build Prefix Array O(n)
vector<int> P(n);
P[0] = A[0];
for (int i = 1; i < n; i++) {
  P[i] = P[i-1] + A[i];
}

// Phase 2: Range Query O(1)
int querySum(int L, int R, vector<int>& P) {
  if (L == 0) return P[R];
  return P[R] - P[L-1];
}
