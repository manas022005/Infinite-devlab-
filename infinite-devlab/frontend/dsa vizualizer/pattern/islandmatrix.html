NUMBER OF ISLANDS

Given a 2D grid of '1's (land) and '0's (water), count the number of islands. An island is surrounded by water and formed by connecting adjacent lands horizontally or vertically.

int numIslands(vector<vector<char>>& grid) {
  int count = 0;
  for (int r = 0; r < ROWS; r++) {
    for (int c = 0; c < COLS; c++) {
      if (grid[r][c] == '1') { // Unvisited land
        count++;
        dfs(grid, r, c); // Flood fill
      }
    }
  }
  return count;
}
void dfs(vector<vector<char>>& grid, int r, int c) {
  if (r < 0 || c < 0 || r >= ROWS || c >= COLS ||
      grid[r][c] != '1') return;
  grid[r][c] = '2'; // Mark visited
  dfs(grid, r + 1, c); // Down
  dfs(grid, r - 1, c); // Up
  dfs(grid, r, c + 1); // Right
  dfs(grid, r, c - 1); // Left
}
