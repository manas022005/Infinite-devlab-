N-QUEENS PUZZLE

Place N non-attacking queens on an N×N chessboard. Each step tries a column in the current row, checks safety across rows, columns, and both diagonals, recurses down, then backtracks when stuck.

solve(row, n)
void solve(int row, int n) {
  if (row == n) { // All queens placed
    solutions.push_back(board);
    return;
  }
  for (int col = 0; col < n; col++) {
    if (isSafe(row, col)) {
      board[row] = col; // Place ♛
      solve(row + 1, n);  // Recurse ↓
      board[row] = -1;  // Backtrack ←
    }
  }
}
isSafe(row, col)
bool isSafe(int row, int col) {
  // Check column ↑
  for (int r = 0; r < row; r++) {
    if (board[r] == col) return false;
  }
  // Check both diagonals ↖ ↗
  for (int r = 0; r < row; r++) {
    if (abs(board[r] - col) == abs(r - row)) {
      return false;
    }
  }
  return true;
}
