void insert(int val) {
  heap.push_back(val);             // Add to end
  heapifyUp(heap.size() - 1);     // Bubble up
}
int extractMax() {
  int max_val = heap[0];            // Root is max
  heap[0] = heap.back(); heap.pop_back();
  heapifyDown(0);                   // Sink down
  return max_val;
}
