void put(string key, int val) {
  int idx = hash(key) % TABLE_SIZE;
  for (auto& node : table[idx]) {        // Traverse Chain
    if (node.key == key) {              // Key exists?
      node.val = val;                   // Update value
      return;
    }
  }
  table[idx].push_back({key, val});    // Insert at end
}
