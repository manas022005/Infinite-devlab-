STACK OPERATIONS

Stack follows the Last-In-First-Out (LIFO) principle. You can only insert (push) or remove (pop) elements from the top.

#include <stack>

int main() {
  stack<int> s;

  s.push(10); // Push to top
  s.push(20);
  s.push(30);
  s.push(40);

  s.pop();  // Remove from top
  s.pop();

  int top_val = s.top(); // Peek
  return 0;
}
#include <queue>

int main() {
  queue<int> q;

  q.push(10); // Enqueue at back
  q.push(20);
  q.push(30);
  q.push(40);

  q.pop();  // Dequeue from front
  q.pop();

  int front_val = q.front(); // Peek
  return 0;
}
